@startuml

class Item{
    Vec3 position
    volume() : int
}

class Vec3

abstract Packer {
    setLimits()
    setPolicy()
    intersectsAny(Item toPack) : bool
    fitsWithinLimits(Item toPack) : bool
    volume() : int
    usedVolume() : int
    {abstract} pack(Item toPack)
}

class GreedyPacker {
    pack(Item toPack)
}

class ExtremePointsPacker {
    pack(Item toPack)
}

class ShelfPacker {
    pack(Item toPack)
}

class Shelf2D {}

interface IRotationPolicy {
    {abstract} rotateBin(Item toPack, Vec3 binSize)
}

class RP_largestFaceUp
class RP_minLeftoverSlack
class RP_tryFirstFitting

Packer <|--GreedyPacker
Packer <|--ExtremePointsPacker
Packer <|--ShelfPacker
ShelfPacker *-- Shelf2D
GreedyPacker *-- IRotationPolicy
IRotationPolicy <|-- RP_largestFaceUp
IRotationPolicy <|-- RP_minSumLeftoverSlack
IRotationPolicy <|-- RP_tryFirstFitting

@enduml
